import tkinter as tk
from dataclasses import dataclass
from tkinter import messagebox
from typing import Optional


BOARD_SIZE = 10

LAKES = {
    (4, 2), (4, 3), (5, 2), (5, 3),
    (4, 6), (4, 7), (5, 6), (5, 7),
}

RED = "Rot"
BLUE = "Blau"

PIECE_COUNTS = {
    "10": 1,  # Feldmarschall
    "9": 1,   # General
    "8": 2,   # Oberst
    "7": 3,   # Major
    "6": 4,   # Hauptmann
    "5": 4,   # Leutnant
    "4": 4,   # Unteroffizier
    "3": 5,   # Mineur
    "2": 8,   # Aufklärer
    "1": 1,   # Spion
    "B": 6,   # Bombe
    "F": 1,   # Fahne
}

PIECE_NAMES = {
    "10": "Feldmarschall",
    "9": "General",
    "8": "Oberst",
    "7": "Major",
    "6": "Hauptmann",
    "5": "Leutnant",
    "4": "Unteroffizier",
    "3": "Mineur",
    "2": "Aufklärer",
    "1": "Spion",
    "B": "Bombe",
    "F": "Fahne",
}


@dataclass
class Piece:
    owner: str
    rank: str

    @property
    def movable(self) -> bool:
        return self.rank not in {"B", "F"}

    @property
    def name(self) -> str:
        return PIECE_NAMES[self.rank]


class StrategoGame:
    def __init__(self) -> None:
        self.board: list[list[Optional[Piece]]] = [
            [None for _ in range(BOARD_SIZE)] for _ in range(BOARD_SIZE)
        ]
        self.current_player = RED
        self.game_over = False
        self.winner: Optional[str] = None
        self.setup_random_board()

    def setup_random_board(self) -> None:
        self.board = [[None for _ in range(BOARD_SIZE)] for _ in range(BOARD_SIZE)]

        red_pieces = self._create_piece_set(RED)
        blue_pieces = self._create_piece_set(BLUE)

        random.shuffle(red_pieces)
        random.shuffle(blue_pieces)

        index = 0
        for row in range(6, 10):
            for col in range(BOARD_SIZE):
                self.board[row][col] = red_pieces[index]
                index += 1

        index = 0
        for row in range(0, 4):
            for col in range(BOARD_SIZE):
                self.board[row][col] = blue_pieces[index]
                index += 1

    @staticmethod
    def _create_piece_set(owner: str) -> list[Piece]:
        pieces: list[Piece] = []
        for rank, count in PIECE_COUNTS.items():
            for _ in range(count):
                pieces.append(Piece(owner, rank))
        return pieces

    def valid_moves_for(self, row: int, col: int) -> list[tuple[int, int]]:
        piece = self.board[row][col]
        if piece is None or piece.owner != self.current_player or not piece.movable:
            return []

        if piece.rank == "2":
            return self._scout_moves(row, col)

        moves = []
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            target = row + dr, col + dc
            if self.is_legal_move((row, col), target):
                moves.append(target)
        return moves

    def _scout_moves(self, row: int, col: int) -> list[tuple[int, int]]:
        moves = []

        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            current_row = row + dr
            current_col = col + dc

            while self._inside_board(current_row, current_col):
                if (current_row, current_col) in LAKES:
                    break

                target = self.board[current_row][current_col]

                if target is None:
                    moves.append((current_row, current_col))
                else:
                    if target.owner != self.current_player:
                        moves.append((current_row, current_col))
                    break

                current_row += dr
                current_col += dc

        return moves

    def is_legal_move(self, source: tuple[int, int], target: tuple[int, int]) -> bool:
        source_row, source_col = source
        target_row, target_col = target

        if not self._inside_board(target_row, target_col):
            return False

        if target in LAKES:
            return False

        piece = self.board[source_row][source_col]
        if piece is None:
            return False

        if piece.owner != self.current_player:
            return False

        if not piece.movable:
            return False

        target_piece = self.board[target_row][target_col]
        if target_piece is not None and target_piece.owner == piece.owner:
            return False

        row_distance = abs(target_row - source_row)
        col_distance = abs(target_col - source_col)

        if row_distance > 0 and col_distance > 0:
            return False

        if piece.rank != "2":
            return row_distance + col_distance == 1

        return self._is_clear_straight_line(source, target)

    def _is_clear_straight_line(
        self,
        source: tuple[int, int],
        target: tuple[int, int],
    ) -> bool:
        source_row, source_col = source
        target_row, target_col = target

        row_step = 0
        col_step = 0

        if target_row > source_row:
            row_step = 1
        elif target_row < source_row:
            row_step = -1

        if target_col > source_col:
            col_step = 1
        elif target_col < source_col:
            col_step = -1

        current_row = source_row + row_step
        current_col = source_col + col_step

        while (current_row, current_col) != (target_row, target_col):
            if (current_row, current_col) in LAKES:
                return False
            if self.board[current_row][current_col] is not None:
                return False
            current_row += row_step
            current_col += col_step

        return True

    def move(self, source: tuple[int, int], target: tuple[int, int]) -> str:
        if not self.is_legal_move(source, target):
            return "Ungültiger Zug."

        source_row, source_col = source
        target_row, target_col = target

        attacker = self.board[source_row][source_col]
        defender = self.board[target_row][target_col]

        if attacker is None:
            return "Ungültiger Zug."

        if defender is None:
            self.board[target_row][target_col] = attacker
            self.board[source_row][source_col] = None
            result = f"{attacker.name} wurde bewegt."
        else:
            result = self._resolve_attack(source, target, attacker, defender)

        if not self.game_over:
            self.current_player = BLUE if self.current_player == RED else RED
            self._check_no_moves_available()

        return result

    def _resolve_attack(
        self,
        source: tuple[int, int],
        target: tuple[int, int],
        attacker: Piece,
        defender: Piece,
    ) -> str:
        source_row, source_col = source
        target_row, target_col = target

        if defender.rank == "F":
            self.board[target_row][target_col] = attacker
            self.board[source_row][source_col] = None
            self.game_over = True
            self.winner = attacker.owner
            return f"{attacker.owner} gewinnt! Die Fahne wurde erobert."

        attacker_wins = False
        both_removed = False

        if defender.rank == "B":
            if attacker.rank == "3":
                attacker_wins = True
            else:
                attacker_wins = False
        elif attacker.rank == "1" and defender.rank == "10":
            attacker_wins = True
        elif attacker.rank == defender.rank:
            both_removed = True
        else:
            attacker_wins = int(attacker.rank) > int(defender.rank)

        if both_removed:
            self.board[source_row][source_col] = None
            self.board[target_row][target_col] = None
            return (
                f"Beide Figuren wurden geschlagen: "
                f"{attacker.name} gegen {defender.name}."
            )

        if attacker_wins:
            self.board[target_row][target_col] = attacker
            self.board[source_row][source_col] = None
            return f"{attacker.name} schlägt {defender.name}."

        self.board[source_row][source_col] = None
        return f"{attacker.name} verliert gegen {defender.name}."

    def _check_no_moves_available(self) -> None:
        if self.has_any_legal_move(self.current_player):
            return

        self.game_over = True
        self.winner = BLUE if self.current_player == RED else RED

    def has_any_legal_move(self, player: str) -> bool:
        original_player = self.current_player
        self.current_player = player

        try:
            for row in range(BOARD_SIZE):
                for col in range(BOARD_SIZE):
                    piece = self.board[row][col]
                    if piece is not None and piece.owner == player and piece.movable:
                        if self.valid_moves_for(row, col):
                            return True
            return False
        finally:
            self.current_player = original_player

    @staticmethod
    def _inside_board(row: int, col: int) -> bool:
        return 0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE

    def __init__(self, root: tk.Tk) -> None:

        for row in range(BOARD_SIZE):
            for col in range(BOARD_SIZE):
                button = self.buttons[row][col]

                        text="≈",

                        fg="white",

                    bg = "#dbeafe"

                    bg = "#fee2e2"

                    bg = "#facc15"

                    bg = "#86efac"

                else:
                    text = "?"
                    fg = "#111827"

                button.config(
                    text=text,
                    bg=bg,
                    fg=fg,
                    state=tk.NORMAL,
                )

    def _on_cell_clicked(self, row: int, col: int) -> None:
        if self.game.game_over:
            return

        if self.selected is not None and (row, col) in self.valid_targets:
            result = self.game.move(self.selected, (row, col))
            self.selected = None
            self.valid_targets = []
            self._refresh_board()

            if self.game.game_over:
                self._show_game_over(result)
            else:
                messagebox.showinfo("Zug Ergebnis", result)
                self._show_cover_window()
                self._refresh_board()

                    "Keine Züge",
                    f"{piece.name} hat aktuell keinen gültigen Zug.",
                )

            self._refresh_board()
            return

        self.valid_targets = []
        self._refresh_board()

    def _show_game_over(self, result: str) -> None:
        self._refresh_board()

        if self.game.winner is None:
            message = result
        else:

        self.selected = None

        self._show_cover_window(initial=True)

        return moves

    def is_legal_move(self, source: tuple[int, int], target: tuple[int, int]) -> bool:
        source_row, source_col = source
        target_row, target_col = target

        if not self._inside_board(target_row, target_col):
            return False

        if target in LAKES:
            return False

        piece = self.board[source_row][source_col]
        if piece is None:
            return False

        if piece.owner != self.current_player:
            return False

        if not piece.movable:
            return False

        target_piece = self.board[target_row][target_col]
        if target_piece is not None and target_piece.owner == piece.owner:
            return False

        row_distance = abs(target_row - source_row)
        col_distance = abs(target_col - source_col)

        if row_distance > 0 and col_distance > 0:
            return False

        if piece.rank != "2":
            return row_distance + col_distance == 1

        return self._is_clear_straight_line(source, target)

    def _is_clear_straight_line(
        self,
        source: tuple[int, int],
        target: tuple[int, int],
    ) -> bool:
        source_row, source_col = source
        target_row, target_col = target

        row_step = 0
        col_step = 0

        if target_row > source_row:
            row_step = 1
        elif target_row < source_row:
            row_step = -1

        if target_col > source_col:
            col_step = 1
        elif target_col < source_col:
            col_step = -1

        current_row = source_row + row_step
        current_col = source_col + col_step

        while (current_row, current_col) != (target_row, target_col):
            if (current_row, current_col) in LAKES:
                return False
            if self.board[current_row][current_col] is not None:
                return False
            current_row += row_step
            current_col += col_step

        return True

    def move(self, source: tuple[int, int], target: tuple[int, int]) -> str:
        if not self.is_legal_move(source, target):
            return "Ungültiger Zug."

        source_row, source_col = source
        target_row, target_col = target

        attacker = self.board[source_row][source_col]
        defender = self.board[target_row][target_col]

        if attacker is None:
            return "Ungültiger Zug."

        if defender is None:
            self.board[target_row][target_col] = attacker
            self.board[source_row][source_col] = None
            result = f"{attacker.owner}: {attacker.name} wurde bewegt."
        else:
            result = self._resolve_attack(source, target, attacker, defender)

        if not self.game_over:
            self.current_player = BLUE if self.current_player == RED else RED
            self._check_no_moves_available()

        return result

    def _resolve_attack(
        self,
        source: tuple[int, int],
        target: tuple[int, int],
        attacker: Piece,
        defender: Piece,
    ) -> str:
        source_row, source_col = source
        target_row, target_col = target

        if defender.rank == "F":
            self.board[target_row][target_col] = attacker
            self.board[source_row][source_col] = None
            self.game_over = True
            self.winner = attacker.owner
            return (
                f"{attacker.owner} gewinnt!\n\n"
                f"{attacker.name} ({attacker.rank}) erobert die Fahne."
            )

        attacker_wins = False
        both_removed = False

        if defender.rank == "B":
            if attacker.rank == "3":
                attacker_wins = True
            else:
                attacker_wins = False
        elif attacker.rank == "1" and defender.rank == "10":
            attacker_wins = True
        elif attacker.rank == defender.rank:
            both_removed = True
        else:
            attacker_wins = int(attacker.rank) > int(defender.rank)

        if both_removed:
            self.board[source_row][source_col] = None
            self.board[target_row][target_col] = None
            return (
                "Kampf unentschieden.\n\n"
                f"{attacker.owner}: {attacker.name} ({attacker.rank})\n"
                f"gegen\n"
                f"{defender.owner}: {defender.name} ({defender.rank})\n\n"
                "Beide Figuren werden entfernt."
            )

        if attacker_wins:
            self.board[target_row][target_col] = attacker
            self.board[source_row][source_col] = None
            return (
                "Angreifer gewinnt.\n\n"
                f"{attacker.owner}: {attacker.name} ({attacker.rank})\n"
                f"schlägt\n"
                f"{defender.owner}: {defender.name} ({defender.rank})"
            )

        self.board[source_row][source_col] = None
        return (
            "Verteidiger gewinnt.\n\n"
            f"{attacker.owner}: {attacker.name} ({attacker.rank})\n"
            f"verliert gegen\n"
            f"{defender.owner}: {defender.name} ({defender.rank})"
        )

    def _check_no_moves_available(self) -> None:
        if self.has_any_legal_move(self.current_player):
            return

        self.game_over = True
        self.winner = BLUE if self.current_player == RED else RED

    def has_any_legal_move(self, player: str) -> bool:
        original_player = self.current_player
        self.current_player = player

        try:
            for row in range(BOARD_SIZE):
                for col in range(BOARD_SIZE):
                    piece = self.board[row][col]
                    if piece is not None and piece.owner == player and piece.movable:
                        if self.valid_moves_for(row, col):
                            return True
            return False
        finally:
            self.current_player = original_player

    @staticmethod
    def _inside_board(row: int, col: int) -> bool:
        return 0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE
